  def disability_to_dummy(x):
    """
    Read a column with multiple options of disability.
    
    Parameters
    ----------
    x: string
       a disability or not.
       
    Returns
    -------
    If there is a disability return 1, else return 0.
    """
    if (x == "NINGUNA" or
        x == "SE IGNORA"):
        return 0
    else:
        return 1
